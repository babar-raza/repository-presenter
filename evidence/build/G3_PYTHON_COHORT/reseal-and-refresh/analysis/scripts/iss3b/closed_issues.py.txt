import json, os, subprocess, sys
sp = sys.argv[1]
h = json.load(open(sp + "/heads_issues.json"))
tok = subprocess.run(["gh","auth","token"],capture_output=True,text=True).stdout.strip()
env = dict(os.environ, GH_TOKEN=tok)
out = {}
for repo in h:
    r = subprocess.run(["gh","api",f"repos/{repo}/issues?state=closed&per_page=100","--paginate","--jq",".[] | select(.pull_request|not) | {n:.number,t:.title,u:.user.login,c:.created_at[:10],x:.closed_at[:10],r:.state_reason}"],capture_output=True,text=True,encoding="utf-8",errors="replace",env=env)
    items = [json.loads(l) for l in r.stdout.splitlines() if l.strip()]
    out[repo] = items
    if items:
        print(repo.split("/")[1])
        for i in items: print("   #%d %s [%s] %s -> %s %s" % (i["n"], i["t"][:90], i["u"], i["c"], i["x"], i["r"]))
json.dump(out, open(sp + "/closed_issues.json","w"), indent=1)
