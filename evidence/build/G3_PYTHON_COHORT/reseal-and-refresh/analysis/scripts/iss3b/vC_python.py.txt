def venv(name):
    p = VEN / name
    if not (p / "Scripts" / "python.exe").exists():
        sh(f'"{PYEXE}" -m venv "{p}"', timeout=300)
    return str(p / "Scripts" / "python.exe").replace("\\", "/")


def fence_lines(clone, start, end):
    lines = (CLONES / clone / "README.md").read_text(encoding="utf-8").splitlines()[start - 1:end]
    return "\n".join(lines) + "\n"


def fence_at(clone, line_no):
    txt = (CLONES / clone / "README.md").read_text(encoding="utf-8").splitlines()
    i = line_no - 1
    while not txt[i].startswith("```python"):
        i -= 1
    j = i + 1
    while not txt[j].startswith("```"):
        j += 1
    return "\n".join(txt[i + 1:j]) + "\n", i + 2, j
