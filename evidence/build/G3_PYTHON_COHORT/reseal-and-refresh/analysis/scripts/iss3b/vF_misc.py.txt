def venv(name):
    p = VEN / name
    if not (p / "Scripts" / "python.exe").exists():
        sh(f'"{PYEXE}" -m venv "{p}"', timeout=300)
    return str(p / "Scripts" / "python.exe").replace("\\", "/")


def case_page_pypi():
    py = venv("3dpy")
    L = Log("metadata-url-Page-Python", "Aspose.Page-FOSS-for-Python")
    L.run("git show HEAD:pyproject.toml | grep -n -i 'url' ; echo grep-exit=$?")
    L.run("git show HEAD:setup.py | grep -n -i 'url' ; echo grep-exit=$?")
    L.run("git show HEAD:pyproject.toml | grep -c '^\\[project.urls\\]'")
    L.run(f'curl -s https://pypi.org/pypi/aspose-page-foss/json | "{py}" -c "import json,sys; i=json.load(sys.stdin)[\'info\']; print(i[\'version\'], i[\'home_page\'], i[\'project_urls\'])"')
    L.save()
