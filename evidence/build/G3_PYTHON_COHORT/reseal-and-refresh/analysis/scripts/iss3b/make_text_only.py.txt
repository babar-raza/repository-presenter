"""Draft the verified findings the handoff model cannot carry, and hold each text to the quality gate.

The text is held to the same gate as a handoff (issue_quality_findings, through a throwaway Handoff that
is never written); the output is a reviewed markdown file the owner can paste by hand.
"""
import subprocess
import sys
from pathlib import Path

from repository_presenter.components.issues.model import EvidenceEntry, Handoff, TriggeringCheck
from repository_presenter.components.issues.quality import issue_quality_findings

CLONES = Path(sys.argv[1])
OUT = Path(sys.argv[2])


def head(name):
    return subprocess.run(["git", "-C", str(CLONES / name), "rev-parse", "HEAD"], capture_output=True, text=True).stdout.strip()


def fence(text):
    return "```\n" + text.strip("\n") + "\n```"


ITEMS = []


def item(*, key, clone, repo, branch, env, title, summary, steps, expected, actual, where, fix, row):
    rev = head(clone)
    body = "\n".join(
        [
            "## Summary", "", summary.strip(), "",
            "## Environment", "",
            f"- Repository: {repo}",
            f"- Revision: `{rev}` (the upstream revision this report was verified against, default branch `{branch}`)",
            *[f"- {line}" for line in env],
            "- Verified on 2026-10-11 from a fresh shallow clone.", "",
            "## Steps to reproduce", "", fence(steps), "",
            "## Expected", "", expected.strip(), "",
            "## Actual", "", actual.strip(), "",
            "## Where in the repository", "", *[f"- {line}" for line in where], "",
            "## Suggested fix", "", fix.strip(), "",
        ]
    )
    handoff = Handoff(
        schema_version=1, repository=repo, source_revision=rev,
        defect_fingerprint="sha256:" + "0" * 64,
        triggering_check=TriggeringCheck(id="BC-02", version="5", causal_stage="EXTRACTING"),
        evidence=(EvidenceEntry(path="x", detail=None),), claim="x",
        suggested_issue_title=title, suggested_issue_body=body, status="HANDOFF_PENDING",
        issue_ref=None, close_reason=None,
    )
    findings = issue_quality_findings(handoff)
    if findings:
        raise SystemExit(f"{key}: " + "; ".join(str(f) for f in findings))
    ITEMS.append((key, row, repo, rev, title, body))
