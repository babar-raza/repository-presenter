open_issues = {r: [i for i in row.get("open_issues", []) if not i["is_pr"]] for r, row in heads.items()}


def dup_text(repo, extra):
    if extra:
        return extra
    oi = open_issues.get(repo, [])
    if not oi:
        return "none: no open issue in the repository"
    return "no overlap with open " + "; ".join(f"#{i['number']} {i['title'][:50]}" for i in oi)


def blockers_text(r):
    out = []
    for b in r["blockers"]:
        if b.startswith("registry write gate"):
            out.append("registry (dry_run: needs issues_mode file)")
        elif b.startswith("owner approval"):
            out.append("approval (no record)")
        elif b.startswith("not independently confirmed"):
            out.append("re-verification " + b.split(": ", 1)[1].split(" at ")[0])
        elif b.startswith("issue text fails"):
            out.append("quality gate")
        elif b.startswith("status is"):
            out.append("status FILED")
        else:
            out.append(b[:60])
    return ", ".join(out)

        "wave": wave,
        "handoff_id": hid,
        "digest": r["evidence_digest"],
        "kind": "handoff",
    })
