def venv(name):
    p = VEN / name
    if not (p / "Scripts" / "python.exe").exists():
        sh(f'"{PYEXE}" -m venv "{p}"', timeout=300)
    return str(p / "Scripts" / "python.exe").replace("\\", "/")


def case_cellspy():
    clone = "Aspose.Cells-FOSS-for-Python"
    py = venv("cellspy")
    w = WORK / "cellspy"

    w.mkdir()

    L.run("git log -3 --format='%h %an %ad %s' --date=short; git show --stat --format= HEAD | tail -4")
