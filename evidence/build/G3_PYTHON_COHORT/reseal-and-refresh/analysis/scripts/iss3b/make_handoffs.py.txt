"""Draft the new hand-authored upstream-defect handoffs through the project's model and quality gate.

Run from the lane worktree with PYTHONPATH=src. Writes under evidence/upstream-defects/ of the
worktree; refuses to write any text the quality gate rejects.
"""
import json
import sys
from pathlib import Path

from repository_presenter.components.issues.classes import fingerprint
from repository_presenter.components.issues.draft import handoff_path
from repository_presenter.components.issues.model import (
    EvidenceEntry,
    Handoff,
    TriggeringCheck,
    load_handoff,
    write_handoff,
)
from repository_presenter.components.issues.quality import issue_quality_findings
from repository_presenter.components.issues.redetect import _current_check_version

ROOT = Path(sys.argv[1])  # evidence/upstream-defects
DATE = "2026-10-11"


def fence(text: str) -> str:
    return "```\n" + text.strip("\n") + "\n```"


def build(
    *,
    repo,
    rev,
    branch,
    env_lines,
    summary,
    steps,
    expected,
    actual,
    where,
    fix,
    title,
    check_id,
    defect_class,
    signature,
    evidence,
    claim,
    verified="Verified on 2026-10-11 from a fresh shallow clone.",
):
    fp = fingerprint(repo, check_id, signature)
    body = [
        "## Summary",
        "",
        summary.strip(),
        "",
        "## Environment",
        "",
        f"- Repository: {repo}",
        f"- Revision: `{rev}` (the upstream revision this report was verified against, default branch `{branch}`)",
        *[f"- {line}" for line in env_lines],
        f"- {verified}",
        "",
        "## Steps to reproduce",
        "",
        fence(steps),
        "",
        "## Expected",
        "",
        expected.strip(),
        "",
        "## Actual",
        "",
        actual.strip(),
        "",
        "## Where in the repository",
        "",
        *[f"- {line}" for line in where],
        "",
        "## Suggested fix",
        "",
        fix.strip(),
        "",
        f"<!-- repository-presenter-defect: {fp} -->",
    ]
    handoff = Handoff(
        schema_version=1,
        repository=repo,
        source_revision=rev,
        defect_fingerprint=fp,
        triggering_check=TriggeringCheck(
            id=check_id, version=_current_check_version(check_id) or "1", causal_stage="EXTRACTING"
        ),
        evidence=tuple(EvidenceEntry(path=p, detail=d) for p, d in evidence),
        claim=claim,
        suggested_issue_title=title,
        suggested_issue_body="\n".join(body) + "\n",
        status="HANDOFF_PENDING",
        issue_ref=None,
        close_reason=None,
        defect_class=defect_class,
    )
    findings = issue_quality_findings(handoff)
    if findings:
        raise SystemExit(f"{repo} {title}: quality gate findings: " + "; ".join(str(f) for f in findings))
    return handoff


def write(handoff):
    path = handoff_path(ROOT, handoff)
    write_handoff(handoff, path)
    print("wrote", path.relative_to(ROOT.parent.parent), handoff.defect_fingerprint[7:15], "|", handoff.suggested_issue_title)
    return path
