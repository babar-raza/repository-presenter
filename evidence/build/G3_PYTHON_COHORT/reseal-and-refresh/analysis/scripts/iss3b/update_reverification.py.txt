path = root / "reverification.json"
doc = json.loads(path.read_text(encoding="utf-8"))
old = {e["handoff_id"]: e for e in doc["entries"]}
entries = {}
for owner_dir in sorted(root.iterdir()):
    if not owner_dir.is_dir():
        continue
    for f in sorted(owner_dir.glob("*.json")):
        h = load_handoff(f)
        hid = f"{owner_dir.name}__{f.stem}"
        short = f.stem[:8]
        digest = evidence_digest(h)
        head = heads[h.repository]["head"] if h.repository in heads else None
        prev = old.get(hid)
        if short in R:
            verdict, method, note, logs = R[short]
            base_note = prev["note"] if prev else ""
        elif short in KEEP_VERDICT:
            verdict = prev["verdict"]
            method, logs = EXTRA[short]
            note, base_note = "", prev["note"]
        elif short in NEW:
            verdict, method, logs = NEW[short]
            note, base_note = "", ""
        else:
            if prev:
                entries[hid] = prev  # e.g. nothing else is expected here
            continue
        live = head if verdict != "FIXED_OR_CHANGED" else prev["live_head_sha"]
        if verdict == "NOT_A_DEFECT" and short in {"a6f52c1a", "ef667e7d"}:
            live = prev["live_head_sha"]
        logtxt = ", ".join(f"{RAWLOGS}/iss3b-{k}.txt" for k in logs)
        merged_note = " ".join(x for x in (base_note, note) if x).strip()
        entries[hid] = {
            "handoff_id": hid,
            "verdict": verdict,
            "live_head_sha": live,
            "verified_at": now if verdict != "FIXED_OR_CHANGED" else prev["verified_at"],
            "evidence_digest": digest,
            "method": f"{method}; re-verified {now[:10]} (lane iss3b); raw logs: {logtxt}",
            "note": merged_note or "Re-verified at the live head.",
        }
        if verdict == "FIXED_OR_CHANGED":
            entries[hid]["method"] = f"{prev['method']}; {method}; re-read {now[:10]} (lane iss3b); raw logs: {logtxt}"
            entries[hid]["verified_at"] = now
        if verdict == "NOT_A_DEFECT" and short in {"a6f52c1a", "ef667e7d"}:
            entries[hid]["method"] = f"{prev['method']}; {method}; re-read {now[:10]} (lane iss3b); raw logs: {logtxt}"
            entries[hid]["verified_at"] = now
# keep any previous entry whose handoff file no longer exists out of the record
doc["entries"] = list(entries.values())
path.write_text(json.dumps(doc, indent=2, ensure_ascii=False) + "\n", encoding="utf-8", newline="\n")
print("entries:", len(doc["entries"]))
