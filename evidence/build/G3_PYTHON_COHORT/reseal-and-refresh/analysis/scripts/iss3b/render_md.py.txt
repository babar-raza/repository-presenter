def esc(x):
    return str(x).replace("|", "\\|").replace("\n", " ")


def short(repo):
    return repo.split("/", 1)[-1]


def rows_where(**kw):

def wave_repos(ws):

def tbl_handoff(items):
    lines = [
        "| id | repository | defect class | proposed title (exact) | sev | evidence (raw-logs) | reproduced 2026-10-11 | duplicate check | readiness blockers | action |",
        "|---|---|---|---|---|---|---|---|---|---|",
    ]
    for t in items:
        wave = f" wave {t['wave']}" if t["wave"] != "-" else ""
        lines.append(
            f"| {t['id']} | {short(t['repo'])} ({t['mode']}) | {esc(t['class'])} | {esc(t['title'])} | {t['severity']} | {esc(t['evidence'].replace('raw-logs/', ''))} | {esc(t['reproduced'])} | {esc(t['duplicate'])} | {esc(t['blockers'])} | **{t['action']}**{wave} |"
        )
    return "\n".join(lines)


def tbl_text(items):
    lines = [
        "| id | repository | defect class | proposed title | sev | evidence (raw-logs) | reproduced 2026-10-11 | duplicate check | readiness blockers | action |",
        "|---|---|---|---|---|---|---|---|---|---|",
    ]
    for t in items:
        lines.append(
            f"| {t['id']} | {esc(short(t['repo']))} ({t['mode']}) | {esc(t['class'])} | {esc(t['title'])} | {t['severity']} | {esc(t['evidence'].replace('raw-logs/', ''))} | {esc(t['reproduced'])} | {esc(t['duplicate'])} | {esc(t['blockers'])} | **{t['action']}** |"
        )
    return "\n".join(lines)
