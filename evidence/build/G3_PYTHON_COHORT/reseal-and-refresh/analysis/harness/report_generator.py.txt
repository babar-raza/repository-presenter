def table(rows, cols, heads=None):
    heads = heads or cols
    s = "| " + " | ".join(heads) + " |\n|" + "|".join("---" for _ in cols) + "|\n"
    for r in rows:
        s += "| " + " | ".join(str(r[c]).replace("|", "/") for c in cols) + " |\n"
    return s
